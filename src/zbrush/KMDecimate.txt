// ---------------------------------------------------------------- subtool copy, export, import
// duplicate the active subtool and make the copy active (a closed folder is opened around the duplicate, as ZCY does)
[RoutineDef,kdDuplicate,
	[VarSet,kdIdx,[SubToolGetActiveIndex]]
	[VarSet,kdCF,[SubToolGetFolderIndex]]
	[If,[SubToolGetStatus,kdCF]&1024==1024,[VarSet,kdOpen,0],[VarSet,kdOpen,1]]
	[If,(kdCF>-1)&&(kdOpen==0),[SubToolSetStatus,kdCF,[SubToolGetStatus,kdCF]+0x400]]
	[IPress,Tool:SubTool:Duplicate]
	[If,(kdCF>-1)&&(kdOpen==0),[SubToolSetStatus,kdCF,[SubToolGetStatus,kdCF]-0x400]]
	[If,[SubToolGetActiveIndex]==kdIdx,[SubToolSelect,kdIdx+1]]
]

// the GoZ template lists the records to write: points, faces, polypaint, groups (no UV / mask / crease)
[RoutineDef,kdExport,
	[MemDelete,kdTplM]
	[MemCreateFromFile,kdTplM,[FileNameResolvePath,kdTpl]]
	[MemResize,kdTplM,[MemGetSize,kdTplM]+20,0]
	[FileNameSetNext,kdGoz,MemBlock:kdTplM]
	[IPress,Tool:Export]
	[MemDelete,kdTplM]
]

// Tool:Import may re-unify the export scale / offset of the tool: put them back and import again (as the GoZ plugin does)
[RoutineDef,kdImport,
	[If,[IExists,Tool:Export:Scale],
		[VarSet,kdSc(0),[IGet,Tool:Export:Scale]]
		[VarSet,kdSc(1),[IGet,"Tool:Export:X Offset"]]
		[VarSet,kdSc(2),[IGet,"Tool:Export:Y Offset"]]
		[VarSet,kdSc(3),[IGet,"Tool:Export:Z Offset"]]
	]
	[FileNameSetNext,kdRes]
	[IPress,Tool:Import]
	[If,[IExists,Tool:Export:Scale],
		[VarSet,kdReimp,0]
		[If,kdSc(0)!=[IGet,Tool:Export:Scale],[VarSet,kdReimp,1]]
		[If,kdSc(1)!=[IGet,"Tool:Export:X Offset"],[VarSet,kdReimp,1]]
		[If,kdSc(2)!=[IGet,"Tool:Export:Y Offset"],[VarSet,kdReimp,1]]
		[If,kdSc(3)!=[IGet,"Tool:Export:Z Offset"],[VarSet,kdReimp,1]]
		[If,kdReimp,
			[ISet,Tool:Export:Scale,kdSc(0)]
			[ISet,"Tool:Export:X Offset",kdSc(1)]
			[ISet,"Tool:Export:Y Offset",kdSc(2)]
			[ISet,"Tool:Export:Z Offset",kdSc(3)]
			[FileNameSetNext,kdRes]
			[IPress,Tool:Import]
		]
	]
]

// wait for status.txt written by KMDecimate.exe; meanwhile progress.bin drives the note bar (Esc is caught by the exe:
// ZScript cannot read keys while it waits). No fixed time limit: give up when the heartbeat stops for about 30 s
// (2 minutes before the first one, because an antivirus may hold a freshly installed exe)
[RoutineDef,kdWait,
	[VarSet,kdStat,""]
	[VarSet,kdTmp,[StrMerge,kdWork,"status.txt"]]
	[VarSet,kdPLast,-1]
	[VarSet,kdPT,-1]
	[VarSet,kdPS,0]
	[VarSet,kdPF,0]
	[VarSet,kdIdle,0]
	[Loop,1000000,
		[If,[FileExists,kdTmp],
			[MemDelete,kdStM]
			[MemCreateFromFile,kdStM,kdTmp]
			[MemReadString,kdStM,kdStat]
			[MemDelete,kdStM]
			[LoopExit]
		]
		[If,[FileExists,kdPrg],
			[MemDelete,kdPrM]
			[MemCreateFromFile,kdPrM,kdPrg]
			[If,[MemGetSize,kdPrM]>=12,
				[MemRead,kdPrM,kdPS,0,0]
				[MemRead,kdPrM,kdPF,0,4]
				[MemRead,kdPrM,kdPT,0,8]
			]
			[MemDelete,kdPrM]
		]
		[If,kdPT!=kdPLast,[VarSet,kdPLast,kdPT][VarSet,kdIdle,0],[VarSet,kdIdle,kdIdle+1]]
		[If,(kdPLast<0)&&(kdIdle>1200),[LoopExit]]
		[If,(kdPLast>=0)&&(kdIdle>300),[LoopExit]]
		[If,kdPS==1,
			[If,kdPF>0,
				[NoteBar,[StrMerge,kdWDec,INT(kdPF*100),"%",kdWEsc],kdPF],
				[NoteBar,[StrMerge,kdWPrep,kdWEsc],0.01]
			],
			[If,kdPS==2,
				[NoteBar,[StrMerge,kdWSave,kdWEsc],1],
				[NoteBar,[StrMerge,kdWLoad,kdWEsc],0.01]
			]
		]
		[Delay,0.1]
	]
	[NoteBar,""]
]

// the round trip: (duplicate) -> highest level only -> export GoZ -> KMDecimate.exe -> import the result
[RoutineDef,kdRun,
	[RoutineCall,kdCheckTool]
	[If,kdOK==0,[Exit]]
	[If,![FileExists,[FileNameResolvePath,kdZFU]],[Note,"KM Decimate: ZFileUtils64.dll is missing in ZPlugs64\KMDecimate\",,3][Exit]]
	[If,![FileExists,[FileNameResolvePath,kdExe]],[Note,"KM Decimate: KMDecimate.exe is missing in ZPlugs64\KMDecimate\",,3][Exit]]
	[If,![FileExists,[FileNameResolvePath,kdTpl]],[Note,"KM Decimate: KMDecimate.GoZ is missing in ZPlugs64\KMDecimate\",,3][Exit]]
	[RoutineCall,kdEnsureDirs]
	[RoutineCall,kdGetName]
	[RoutineCall,kdWords]
	[VarSet,kdTmp,kdGoz][RoutineCall,kdDel]
	[VarSet,kdTmp,kdRes][RoutineCall,kdDel]
	[VarSet,kdTmp,[StrMerge,kdWork,"status.txt"]][RoutineCall,kdDel]
	[VarSet,kdTmp,[StrMerge,kdWork,"note.txt"]][RoutineCall,kdDel]
	[VarSet,kdTmp,kdPrg][RoutineCall,kdDel]
	[VarSet,kdStat,""]
	[VarSet,kdCanc,0]
	// a subtool with subdivision levels is always decimated as a copy (deleting its lower levels in place is not undoable)
	[VarSet,kdCopy,kdSet(2)]
	[If,[IGetMax,Tool:Geometry:SDiv]>1,[VarSet,kdCopy,1]]
	[VarSet,kdT0,[ZBrushInfo,2]]
	[RoutineCall,kdPrefsSet]
	[NoteBar,kdWExp,0.01]
	[IFreeze,
		[If,kdCopy,[RoutineCall,kdDuplicate]]
		[If,[IGet,Tool:Geometry:SDiv]<[IGetMax,Tool:Geometry:SDiv],[ISet,Tool:Geometry:SDiv,[IGetMax,Tool:Geometry:SDiv]]]
		[If,kdCopy,[If,[IGetStatus,"Tool:Geometry:Del Lower"],[IPress,"Tool:Geometry:Del Lower"]]]
		[RoutineCall,kdExport]
	]
	[VarSet,kdT1,[ZBrushInfo,2]]
	[VarSet,kdT2,kdT1]
	[VarSet,kdT3,kdT1]
	[If,[FileExists,kdGoz],
		[RoutineCall,kdWriteJob]
		[VarSet,kdTmp,[StrMerge,kdWork,"job.txt"]]
		[RoutineCall,kdLaunch]
		[RoutineCall,kdWait]
		[VarSet,kdT2,[ZBrushInfo,2]]
		[If,([StrLength,kdStat]>0)&&([StrFind,"1",kdStat]==0)&&[FileExists,kdRes],
			[NoteBar,kdWImp,1]
			[IFreeze,
				[RoutineCall,kdImport]
				[If,kdCopy,[RoutineCall,kdRename]]
			]
			[NoteBar,""]
		]
		// Esc in KMDecimate.exe: drop the copy that was made for it and go back to the original
		[If,([StrLength,kdStat]>0)&&([StrFind,"8",kdStat]==0),
			[VarSet,kdCanc,1]
			[If,kdCopy,[IFreeze,[RoutineCall,kdDelete][SubToolSelect,kdIdx]]]
		]
		[VarSet,kdT3,[ZBrushInfo,2]]
	]
	[RoutineCall,kdPrefsRestore]
	[NoteBar,""]
	[If,kdCanc,
		[VarSet,kdMsg,"KM Decimate: cancelled, the model was not changed."][VarSet,kdLI,33][RoutineCall,kdText]
		[Note,kdMsg,,3],
		[If,[FileExists,kdGoz],
			[RoutineCall,kdShowNote],
			[VarSet,kdMsg,"KM Decimate: the GoZ export failed."][VarSet,kdLI,12][RoutineCall,kdText]
			[Note,kdMsg,,3]
		]
	]
	[VarSet,kdTmp,kdGoz][RoutineCall,kdDel]
	[VarSet,kdTmp,kdRes][RoutineCall,kdDel]
	[VarSet,kdTmp,kdPrg][RoutineCall,kdDel]
	[RoutineCall,kdEnableUI]
]

// ---------------------------------------------------------------- remesh: keep the current state as history -> ZRemesher -> subdivide -> project level by level
// Two ways to keep the history:
//   copy (default): duplicate the subtool; the original is the source, the copy is remeshed and gets the details with ProjectAll
//   Project History: works in place; ZBrush projects from the history point the user stored with Ctrl+click on the timeline
//   (ZScript cannot store that point itself)
[RoutineDef,kdSave2,
	[RoutineCall,kdEnsureDirs]
	[VarSave,kdSet2,kdSet2Path]
]

// subdivision levels: fixed, or so that the top level comes closest to the target (each Divide = x4 faces)
[RoutineDef,kdLevels,
	[VarSet,kdTarget,kdP]
	[If,kdSet2(2)>0,[VarSet,kdTarget,kdSet2(2)*10000]]
	[VarSet,kdL,0]
	[If,kdSet2(1)>0,
		[VarSet,kdL,INT(kdSet2(1))],
		[VarSet,kdF,kdR]
		[Loop,8,
			[If,(kdF>0)&&(kdF*2<kdTarget),
				[VarSet,kdF,kdF*4]
				[VarSet,kdL,kdL+1],
				[LoopExit]
			]
		]
	]
]

// ProjectAll projects from every visible subtool: hide all except the source and the result (statuses are put back after)
[RoutineDef,kdHideOthers,
	[VarSet,kdCnt,[SubToolGetCount]]
	[If,kdCnt>1024,[VarSet,kdCnt,1024]]
	[Loop,kdCnt,
		[VarSet,kdHid(kdN),0]
		[If,(kdN!=kdSrc)&&(kdN!=kdDst),
			[VarSet,kdSt(kdN),[SubToolGetStatus,kdN]]
			[If,kdSt(kdN)&1==1,
				[SubToolSetStatus,kdN,kdSt(kdN)-1]
				[VarSet,kdHid(kdN),1]
			]
		]
	,kdN]
	[If,[SubToolGetStatus,kdSrc]&1==0,[SubToolSetStatus,kdSrc,[SubToolGetStatus,kdSrc]+1]]
]

[RoutineDef,kdShowOthers,
	[Loop,kdCnt,
		[If,kdHid(kdN),
			[If,[SubToolGetStatus,kdN]&1==0,[SubToolSetStatus,kdN,[SubToolGetStatus,kdN]+1]]
		]
	,kdN]
]

[RoutineDef,kdProject,
	[If,kdSet2(3),
		[IPress,"Tool:SubTool:Project History"],
		[IPress,Tool:SubTool:ProjectAll]
	]
]

[RoutineDef,kdRemesh,
	[RoutineCall,kdCheckTool]
	[If,kdOK==0,[Exit]]
	[RoutineCall,kdGetName]
	[RoutineCall,kdWords]
	[VarSet,kdT0,[ZBrushInfo,2]]
	[VarSet,kdTPC,[IGet,"Tool:Geometry:Target Polygons Count"]]
	[VarSet,kdSolo,[IGet,Transform:Solo]]
	[VarSet,kdKept,1]
	// the source shows its highest level
	[If,[IGet,Tool:Geometry:SDiv]<[IGetMax,Tool:Geometry:SDiv],[ISet,Tool:Geometry:SDiv,[IGetMax,Tool:Geometry:SDiv]]]
	[Mesh3DGet,1,,,kdP]
	[VarSet,kdSrc,[SubToolGetActiveIndex]]
	[VarSet,kdDst,kdSrc]
	[If,kdSet2(3)==0,
		[IFreeze,[RoutineCall,kdDuplicate]]
		[VarSet,kdDst,[SubToolGetActiveIndex]]
	]
	// one level only (ZRemesher needs that), then remesh to the target
	[If,[IGetMax,Tool:Geometry:SDiv]>1,[If,[IGetStatus,"Tool:Geometry:Del Lower"],[IPress,"Tool:Geometry:Del Lower"]]]
	[VarSet,kdZOK,0]
	[If,[IGetStatus,Tool:Geometry:ZRemesher],
		[NoteBar,kdWZR,0.05]
		[ISet,"Tool:Geometry:Target Polygons Count",kdSet2(0)]
		[IPress,Tool:Geometry:ZRemesher]
		[ISet,"Tool:Geometry:Target Polygons Count",kdTPC]
		[VarSet,kdZOK,1]
	]
	[If,kdZOK==0,
		[NoteBar,""]
		[If,kdSet2(3)==0,[RoutineCall,kdDelete][SubToolSelect,kdSrc]]
		[VarSet,kdMsg,"KM Remesh: ZRemesher is not available for this subtool."][VarSet,kdLI,22][RoutineCall,kdText]
		[Note,kdMsg,,3]
		[RoutineCall,kdEnableUI]
		[Exit]
	]
	[Mesh3DGet,1,,,kdR]
	[RoutineCall,kdLevels]
	[ISet,Transform:Solo,0]
	[If,kdSet2(3)==0,[IFreeze,[RoutineCall,kdHideOthers]]]
	[NoteBar,[StrMerge,kdWDiv,"0 / ",INT(kdL)],0.1]
	[RoutineCall,kdProject]
	[Loop,kdL,
		[NoteBar,[StrMerge,kdWDiv,INT(kdN+1)," / ",INT(kdL)],0.1+0.9*(kdN+1)/(kdL+1)]
		[IPress,Tool:Geometry:Divide]
		[RoutineCall,kdProject]
	,kdN]
	[NoteBar,""]
	[Mesh3DGet,1,,,kdF]
	[If,kdSet2(3)==0,
		[IFreeze,[RoutineCall,kdShowOthers]]
		// the original is deleted only when it is not the top of a folder (deleting that could take the folder with it)
		[If,(kdSet(2)==0)&&([SubToolGetFolderIndex,kdSrc]!=kdSrc),[VarSet,kdKept,0]]
		[If,kdKept,
			[If,[SubToolGetStatus,kdSrc]&1==1,[SubToolSetStatus,kdSrc,[SubToolGetStatus,kdSrc]-1]]
			[VarSet,kdName2,[StrMerge,kdName,"_ZR"]]
			[FileExecute,kdZFU,"RenameSetNext",kdName2]
			[IPress,Tool:SubTool:Rename],
			[SubToolSelect,kdSrc]
			[RoutineCall,kdDelete]
			[SubToolSelect,kdSrc]
			[FileExecute,kdZFU,"RenameSetNext",kdName]
			[IPress,Tool:SubTool:Rename]
		]
	]
	[ISet,Transform:Solo,kdSolo]
	[VarSet,kdT3,[ZBrushInfo,2]]
	[VarSet,kdD3,INT((kdT3-kdT0)*10)/10]
	[VarSet,kdLI,17][VarSet,kdTip,kdW17][RoutineCall,kdTipOf][VarSet,kdW17,kdTip]
	[VarSet,kdLI,18][VarSet,kdTip,kdW18][RoutineCall,kdTipOf][VarSet,kdW18,kdTip]
	[VarSet,kdLI,19][VarSet,kdTip,kdW19][RoutineCall,kdTipOf][VarSet,kdW19,kdTip]
	[VarSet,kdLI,20][VarSet,kdTip,kdW20][RoutineCall,kdTipOf][VarSet,kdW20,kdTip]
	[VarSet,kdLI,21][VarSet,kdTip,kdW21][RoutineCall,kdTipOf][VarSet,kdW21,kdTip]
	[VarSet,kdMsg,[StrMerge,kdW17,INT(kdR),kdW18,INT(kdL),kdW19,INT(kdF),kdW20,INT(kdP),kdW21,kdD3,"s"]]
	[If,(kdSet2(3)==0)&&(kdSet(2)==0)&&kdKept,
		[VarSet,kdTip,"The original is the top of a folder, so it was kept (hidden)."][VarSet,kdLI,23][RoutineCall,kdTipOf]
		[VarSet,kdMsg,[StrMerge,kdMsg,kdNL,kdTip]]
	]
	[Note,kdMsg,,6]
	[RoutineCall,kdEnableUI]
]
